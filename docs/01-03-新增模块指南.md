# 新增模块指南

本文档是面向开发者和 AI 工具的完整操作手册，提供从零新增一个业务模块的全部步骤。以新增一个「文章管理」模块为例，完整演示。

## 业务模块划分规范

项目按业务领域划分目录，避免所有模块堆在 `system` 下：

| 目录        | 业务领域 | 示例模块                           |
| ----------- | -------- | ---------------------------------- |
| `system/`   | 系统管理 | 管理员、角色、菜单、配置、操作日志 |
| `content/`  | 内容管理 | 文章、分类、标签、评论             |
| `business/` | 业务模块 | 订单、商品、客户、合同             |
| `finance/`  | 财务模块 | 发票、付款、对账                   |

## 前置约定

- 资源名: `articles`
- 业务目录: `content`（内容管理）
- 数据库表: `sys_article`
- 权限前缀: `content:article`
- 菜单路径: `/content/article`

## Step 1: 数据库 Schema

### 1.1 创建 Schema 文件

```ts
// src/db/schema/article.ts
import { sql } from "drizzle-orm";
import {
  bigint,
  index,
  mysqlTable,
  text,
  tinyint,
  varchar,
} from "drizzle-orm/mysql-core";
import { localDatetime } from "../custom-types";

export const sysArticle = mysqlTable(
  "sys_article",
  {
    id: bigint("id", { mode: "number", unsigned: true })
      .primaryKey()
      .autoincrement(),
    title: varchar("title", { length: 200 }).notNull(),
    content: text("content"),
    category: varchar("category", { length: 50 }).notNull().default(""),
    status: tinyint("status", { unsigned: true }).notNull().default(1), // 0-草稿 1-发布
    remark: varchar("remark", { length: 500 }),
    createdAt: localDatetime("created_at")
      .notNull()
      .default(sql`CURRENT_TIMESTAMP`),
    updatedAt: localDatetime("updated_at")
      .notNull()
      .default(sql`CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP`),
  },
  (table) => ({
    statusIdx: index("idx_status").on(table.status),
    categoryIdx: index("idx_category").on(table.category),
  }),
);

export type SysArticle = typeof sysArticle.$inferSelect;
export type NewSysArticle = typeof sysArticle.$inferInsert;
```

### 2.2 创建 Service 实现

```ts
// src/server/services/content/article/article.service.ts
import { and, count, eq, like, sql } from "drizzle-orm";
import { db } from "@/db";
import { sysArticle } from "@/db/schema";
import { handleDatabaseError, NotFoundError } from "@/lib/errors";
import {
  buildPaginatedResult,
  normalizePagination,
} from "@/server/utils/pagination";
import type {
  ArticleQuery,
  CreateArticleInput,
  UpdateArticleInput,
} from "@/server/routes/articles/dtos";

export async function getArticleList(query: ArticleQuery) {
  const { page, pageSize, offset } = normalizePagination(query);
  const conditions = [];

  if (query.title) conditions.push(like(sysArticle.title, `%${query.title}%`));
  if (query.status !== undefined)
    conditions.push(eq(sysArticle.status, query.status));
  if (query.category) conditions.push(eq(sysArticle.category, query.category));

  const whereClause = conditions.length > 0 ? and(...conditions) : undefined;

  const [{ total }] = await db
    .select({ total: count() })
    .from(sysArticle)
    .where(whereClause);

  const items = await db
    .select()
    .from(sysArticle)
    .where(whereClause)
    .orderBy(sql`${sysArticle.id} DESC`)
    .limit(pageSize)
    .offset(offset);

  return buildPaginatedResult(items, total, page, pageSize);
}

export async function getArticleById(id: number) {
  const article = await db.query.sysArticle.findFirst({
    where: eq(sysArticle.id, id),
  });
  if (!article) throw new NotFoundError("文章", id);
  return article;
}

export async function createArticle(input: CreateArticleInput) {
  try {
    const [result] = await db.insert(sysArticle).values(input);
    return getArticleById(Number(result.insertId));
  } catch (err) {
    throw handleDatabaseError(err);
  }
}

export async function updateArticle(id: number, input: UpdateArticleInput) {
  await getArticleById(id);
  try {
    await db.update(sysArticle).set(input).where(eq(sysArticle.id, id));
    return getArticleById(id);
  } catch (err) {
    throw handleDatabaseError(err);
  }
}

export async function deleteArticle(id: number) {
  await getArticleById(id);
  await db.delete(sysArticle).where(eq(sysArticle.id, id));
}
```

## Step 3: 服务端 — 路由层

### 3.1 创建 DTOs

```ts
// src/server/routes/articles/dtos.ts
import { z } from "zod";
import { createPaginatedSchema, PaginationQuerySchema } from "../shared";

export const ArticleSchema = z.object({
  id: z.number(),
  title: z.string(),
  content: z.string().nullable(),
  category: z.string(),
  status: z.number(),
  remark: z.string().nullable(),
  createdAt: z.string(),
  updatedAt: z.string(),
});

export const CreateArticleInputSchema = z.object({
  title: z.string().min(1).max(200),
  content: z.string().optional(),
  category: z.string().max(50).optional(),
  status: z.number().int().min(0).max(1).optional(),
  remark: z.string().max(500).optional(),
});

export const UpdateArticleInputSchema = CreateArticleInputSchema.partial();

export const ArticleQuerySchema = PaginationQuerySchema.extend({
  title: z.string().optional(),
  category: z.string().optional(),
  status: z.coerce.number().int().min(0).max(1).optional(),
});

export const PaginatedArticleSchema = createPaginatedSchema(ArticleSchema);

// 类型导出
export type Article = z.infer<typeof ArticleSchema>;
export type CreateArticleInput = z.infer<typeof CreateArticleInputSchema>;
export type UpdateArticleInput = z.infer<typeof UpdateArticleInputSchema>;
export type ArticleQuery = z.infer<typeof ArticleQuerySchema>;
export type PaginatedArticle = z.infer<typeof PaginatedArticleSchema>;
```

### 3.2 创建路由实现

```ts
// src/server/routes/articles/index.ts
import { Hono } from "hono";
// 使用项目封装版本：统一校验错误格式、debug 级别打印原始数据和原因
import { zValidator } from "@/server/utils/validator";
import type { Env } from "@/server/context";
import { requireAuth } from "@/server/middleware/session-auth";
import { requirePermission } from "@/server/middleware/rbac";
import { auditLog } from "@/server/middleware/audit-log";
import { R } from "@/server/utils/response";
import {
  getArticleList,
  getArticleById,
  createArticle,
  updateArticle,
  deleteArticle,
} from "@/server/services";
import { IdParamSchema } from "../shared";
import {
  ArticleQuerySchema,
  CreateArticleInputSchema,
  UpdateArticleInputSchema,
} from "./dtos";

export const articles = new Hono<Env>()
  .use("/*", requireAuth)
  .get(
    "/",
    requirePermission("content:article:list"),
    zValidator("query", ArticleQuerySchema),
    async (c) => R.ok(c, await getArticleList(c.req.valid("query"))),
  )
  .get(
    "/:id",
    requirePermission("content:article:query"),
    zValidator("param", IdParamSchema),
    async (c) => R.ok(c, await getArticleById(c.req.valid("param").id)),
  )
  .post(
    "/",
    requirePermission("content:article:create"),
    auditLog({
      module: "文章管理",
      operation: "创建",
      description: "创建文章",
    }),
    zValidator("json", CreateArticleInputSchema),
    async (c) => R.ok(c, await createArticle(c.req.valid("json"))),
  )
  .put(
    "/:id",
    requirePermission("content:article:update"),
    auditLog({
      module: "文章管理",
      operation: "更新",
      description: "更新文章",
    }),
    zValidator("param", IdParamSchema),
    zValidator("json", UpdateArticleInputSchema),
    async (c) =>
      R.ok(
        c,
        await updateArticle(c.req.valid("param").id, c.req.valid("json")),
      ),
  )
  .delete(
    "/:id",
    requirePermission("content:article:delete"),
    auditLog({
      module: "文章管理",
      operation: "删除",
      description: "删除文章",
    }),
    zValidator("param", IdParamSchema),
    async (c) => {
      await deleteArticle(c.req.valid("param").id);
      return R.success(c, "删除成功");
    },
  );
```

### 3.4 使用 `createCrudMiddlewares` 简化中间件注册（推荐）

对于标准 CRUD 路由，推荐使用 `createCrudMiddlewares` 工厂函数批量生成每个操作的权限检查和审计日志中间件，避免手写重复的 `requirePermission + auditLog` 组合。该工厂函数位于 `src/server/routes/shared/crud/router-factory.ts`。

使用示例：

```ts
import { createCrudMiddlewares } from "@/server/routes/shared/crud/router-factory";

const crudMiddlewares = createCrudMiddlewares({
  moduleName: "文章管理",
  permissionPrefix: "content:article",
});

export const articles = new Hono<Env>()
  .use("/*", requireAuth)
  .get("/", ...crudMiddlewares.list, zValidator("query", ArticleQuerySchema), async (c) => ...)
  .post("/", ...crudMiddlewares.create, zValidator("json", CreateArticleInputSchema), async (c) => ...)
  .put("/:id", ...crudMiddlewares.update, zValidator("param", IdParamSchema), zValidator("json", UpdateArticleInputSchema), async (c) => ...)
  .delete("/:id", ...crudMiddlewares.delete, zValidator("param", IdParamSchema), async (c) => ...);
```

工厂函数会按照约定自动生成各操作对应的权限标识（`{前缀}:list`、`{前缀}:create` 等）及审计日志配置。

### 3.5 自定义 `zValidator` 说明

路由中 **必须** 使用项目封装的 `zValidator`，而非直接使用 `@hono/zod-validator`：

```ts
// ✅ 正确
import { zValidator } from "@/server/utils/validator";

// ❌ 错误 — 不要直接导入原始包
import { zValidator } from "@hono/zod-validator";
```

项目封装版本的作用：
- 将校验失败统一转换为 `ValidationError`，格式与全局错误处理器对齐
- 在 debug 级别打印原始请求数据和校验失败原因，便于本地快速排查参数问题

### 4.1 创建 Hooks 文件

**目录结构建议**：`src/hooks/queries/resources/articles.ts`

```ts
// src/hooks/queries/resources/articles.ts
import { getApiClient } from "@/lib/client";
import type {
  Article,
  ArticleQuery,
  CreateArticleInput,
  PaginatedArticle,
  UpdateArticleInput,
} from "@/server/routes/articles/dtos";
import { createResource } from "../create-resource";

const getClient = () => getApiClient().articles;

const articleResource = createResource<
  PaginatedArticle,
  Article,
  CreateArticleInput,
  UpdateArticleInput,
  ArticleQuery
>({
  resourceName: "articles",
  getClient,
  messages: {
    list: "获取文章列表失败",
    detail: "获取文章详情失败",
    create: "创建文章失败",
    update: "更新文章失败",
    delete: "删除文章失败",
  },
});

export const useArticles = articleResource.useList; // 列表 Hook（复数形式）
export const useArticle = articleResource.useDetail; // 详情 Hook（单数形式）
export const useCreateArticle = articleResource.useCreate;
export const useUpdateArticle = articleResource.useUpdate;
export const useDeleteArticle = articleResource.useDelete;
export const articleKeys = articleResource.keys; // Query Keys（单数 + Keys）
```

## 业务目录创建指南

### 新增业务目录

当需要新增一个业务领域（如订单管理）时：

1. **创建后端服务目录**：

   ```bash
   mkdir -p src/server/services/business/order
   touch src/server/services/business/order/order.service.ts
   touch src/server/services/business/order/index.ts
   ```

2. **创建前端页面目录**：

   ```bash
   mkdir -p src/app/(dashboard)/business/order
   ```

3. **更新导出文件**：
   - `src/server/services/business/index.ts`：添加 `export * from "./order";`
   - `src/server/services/index.ts`：确保导出业务目录

### 目录命名规范

| 目录名      | 用途                         |
| ----------- | ---------------------------- |
| `system/`   | 系统管理类模块，基础架构功能 |
| `content/`  | 内容发布、编辑、审核类功能   |
| `business/` | 核心业务流程，如订单、合同等 |
| `finance/`  | 财务、支付、对账类功能       |
| `member/`   | 会员、客户、用户中心类功能   |
