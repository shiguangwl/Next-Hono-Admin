# 目录结构

## 顶层目录

```
Next-Hono-Admin/
├── docs/                    # 项目文档（本文件所在目录）
├── drizzle/                 # Drizzle 生成的 SQL 迁移文件
├── public/                  # 静态资源
├── script/                  # 工具脚本
├── src/                     # 源代码根目录
│   ├── app/                 # Next.js App Router 页面
│   ├── components/          # 共享 UI 组件
│   ├── db/                  # 数据库层（Schema + 种子 + 连接）
│   ├── hooks/               # React Hooks（包含 React Query 封装）
│   ├── lib/                 # 共享工具库
│   ├── server/              # Hono 后端（路由 + 服务 + 中间件）
│   ├── env.ts               # 环境变量运行时校验
│   └── instrumentation.ts   # Next.js 启动钩子（DB 初始化、监控初始化）
├── middleware.ts             # Next.js Edge Middleware（路由守卫初步检查）
├── next.config.mjs          # Next.js 配置
├── drizzle.config.ts        # Drizzle Kit 配置
├── biome.json               # Biome 代码规范配置
├── vitest.config.ts          # Vitest 测试配置
├── Dockerfile               # 多阶段 Docker 构建
├── docker-compose.yml        # Docker Compose 配置
├── .env.development          # 开发环境变量
└── .env.production           # 生产环境变量模板
```

## `src/app/` — 页面层

```
src/app/
├── layout.tsx               # 根 Layout（MantineProvider、ColorSchemeScript）
├── providers.tsx             # 全局 Providers（Mantine + React Query + Auth + Toaster）
├── globals.css               # 全局样式
├── error.tsx                 # 全局错误边界
├── global-error.tsx          # 全局错误页（完整 HTML）
├── not-found.tsx             # 404 页面
├── icon.svg                  # Favicon
│
├── (auth)/                   # 认证路由组（GuestGuard 保护，已登录则跳转 /dashboard）
│   ├── layout.tsx            # 认证页布局（左侧品牌区 + 右侧表单区）
│   └── login/
│       └── page.tsx          # 登录页
│
├── (dashboard)/              # 管理后台路由组（AuthGuard 保护，未登录则跳转 /login）
│   ├── layout.tsx            # 后台布局（AppShell: Header + Sidebar + Main）
│   ├── dashboard/
│   │   └── page.tsx          # 仪表盘首页
│   └── system/               # 系统管理模块
│       ├── admin/            # 管理员管理
│       ├── role/             # 角色管理
│       ├── menu/             # 菜单管理
│       ├── config/           # 系统配置
│       └── log/              # 操作日志
│
├── (public)/                 # 公开路由组（无需认证）
│
└── api/
    └── [[...route]]/
        └── route.ts          # Hono API 入口（将所有 /api/* 请求转发给 Hono）
```

## `src/server/` — 后端层

```
src/server/
├── app.ts                   # Hono 应用入口（组装中间件 + 路由 + 错误处理）
├── context.ts               # Hono Context 类型定义（Variables 扩展）
├── route-defs.ts            # 路由类型导出（AppType，供 RPC Client 使用）
├── types.ts                 # 服务端类型导出
│
├── config/
│   └── public-paths.ts      # 公开 API 路径配置（跳过 JWT 认证）
│
├── setup/                   # 应用初始化模块
│   ├── middlewares.ts        # 中间件注册（顺序严格固定）
│   ├── routes.ts             # 路由挂载
│   ├── error-handlers.ts     # 全局错误处理器
│   └── audit.ts              # 审计日志初始化
│
├── middleware/               # 中间件实现
│   ├── index.ts              # 统一导出
│   ├── request-context.ts    # 请求上下文（生成 requestId + AsyncLocalStorage）
│   ├── request-logger.ts     # 请求日志（Pino 结构化输出）
│   ├── jwt-auth.ts           # JWT 认证（解析 Bearer Token）
│   ├── rbac.ts               # RBAC 权限（requirePermission 系列）
│   ├── audit-log.ts          # 审计日志中间件
│   ├── rate-limit.ts         # 速率限制（API 全局 + 登录专用）
│   ├── cors.ts               # CORS 配置（可选）
│   └── csrf.ts               # CSRF 防护（可选）
│
├── routes/                   # API 路由（按资源组织）
│   ├── auth/                 # 认证路由（非 CRUD，手动编写）
│   │   ├── dtos.ts           # 请求/响应 Schema
│   │   └── index.ts          # 路由实现
│   ├── admins/               # 管理员（CRUD + 扩展路由）
│   │   ├── dtos.ts
│   │   └── index.ts
│   ├── roles/                # 角色
│   ├── menus/                # 菜单
│   ├── configs/              # 系统配置
│   ├── operation-logs/       # 操作日志
│   └── shared/               # 共享路由工具
│       ├── crud/             # CRUD 工厂
│       │   ├── router-factory.ts   # createCrudMiddlewares（权限 + 审计中间件）
│       │   ├── types.ts            # CRUD 类型定义
│       │   └── index.ts
│       ├── schemas/          # 共享 Schema
│       │   ├── common.ts     # 通用 Schema（ErrorSchema、IdParamSchema 等）
│       │   ├── pagination.ts # 分页 Schema
│       │   └── index.ts
│       └── index.ts
│
├── services/                 # 业务逻辑层
│   ├── index.ts              # 统一导出
│   └── system/               # 系统管理服务
│       ├── admin/            # 管理员服务
│       │   ├── admin.service.ts
│       │   ├── admin.query.ts       # 数据库查询
│       │   ├── admin-role.service.ts
│       │   ├── admin-role.query.ts
│       │   ├── admin-permission.query.ts
│       │   └── index.ts
│       ├── auth/             # 认证服务
│       ├── role/             # 角色服务
│       ├── menu/             # 菜单服务
│       ├── config/           # 配置服务
│       └── audit/            # 审计日志服务
│
└── utils/                    # 服务端工具
    └── response.ts           # R 统一响应工具类 + contextMiddleware
```

## `src/db/` — 数据库层

```
src/db/
├── index.ts                 # 数据库连接（连接池 + Drizzle 实例 + 自动迁移/种子）
├── seed.ts                  # 种子脚本入口
├── seed-runner.ts           # 种子执行器
├── seed-data.ts             # 种子数据定义
├── custom-types/            # 自定义 Drizzle 字段类型
│   ├── local-datetime.ts    # 本地时区 Datetime 类型
│   └── index.ts
└── schema/                  # Drizzle Schema 定义
    ├── index.ts             # 统一导出
    ├── admin.ts             # sys_admin 管理员表
    ├── role.ts              # sys_role 角色表
    ├── menu.ts              # sys_menu 菜单权限表
    ├── admin-role.ts        # sys_admin_role 管理员-角色关联表
    ├── role-menu.ts         # sys_role_menu 角色-菜单关联表
    ├── operation-log.ts     # sys_operation_log 操作日志表
    ├── system-config.ts     # sys_config 系统配置表
    └── relations.ts         # Drizzle Relations 定义
```
