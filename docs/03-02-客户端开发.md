## API Client（RPC Client）

```ts
// src/lib/client.ts
import { hc } from "hono/client";
import type { AppType } from "@/server/route-defs";

export function createClient(token?: string) {
  const baseUrl = getBaseUrl(); // 内部处理环境判断
  return hc<AppType>(`${baseUrl}/api`, {
    headers: () => {
      const headers: Record<string, string> = {
        "Content-Type": "application/json",
      };
      if (token) headers.Authorization = `Bearer ${token}`;
      return headers;
    },
  });
}

export const getApiClient = () => createClient();

// 响应解包器 — 提取 data 字段，统一错误处理
export async function unwrapApiData<T>(
  response: unknown,
  fallbackMessage: string,
): Promise<T> {
  const res = response as any;
  const payload = await res.json().catch(() => null);

  if (!res.ok) {
    if (payload?.message) throw new Error(payload.message);
    throw new Error(fallbackMessage);
  }

  return payload.data as T;
}
```

**使用方式：**

```ts
// 直接调用（类型安全）
const client = getApiClient();
const res = await client.admins.$get({ query: { page: "1", pageSize: "20" } });
const data = await unwrapApiData(res, "获取管理员列表失败");
```

## React Query Hooks

### 工厂模式：createResource

`createResource` 是核心工厂函数，为每个 CRUD 资源自动生成 5 个 Hook：

```ts
// src/hooks/queries/create-resource.ts
const resource = createResource<TList, TDetail, TCreate, TUpdate, TQuery>({
  resourceName: 'admins',
  getClient: () => getApiClient().admins,
  defaultPageSize: 20,
  messages: {
    list: '获取管理员列表失败',
    detail: '获取管理员详情失败',
    create: '创建管理员失败',
    update: '更新管理员失败',
    delete: '删除管理员失败',
  },
})

// 工厂返回的方法：
resource.useList(params?)    // useQuery — 分页列表
resource.useDetail(id)       // useQuery — 单条详情
resource.useCreate()         // useMutation — 创建（成功后自动 invalidate）
resource.useUpdate()         // useMutation — 更新（成功后自动 invalidate）
resource.useDelete()         // useMutation — 删除（成功后自动 invalidate）
resource.keys                // Query Keys 对象

// 注意：实际导出时会按资源名重命名，例如：
// export const useAdmins = adminResource.useList   // 复数形式（列表）
// export const useAdmin = adminResource.useDetail    // 单数形式（详情）
// export const adminKeys = adminResource.keys        // 单数 + Keys
```

### 只读资源：createReadonlyResource

用于操作日志等只需查询 + 删除的资源：

```ts
const logResource = createReadonlyResource<TList, TQuery>({
  resourceName: "operation-logs",
  getClient: () => getApiClient()["operation-logs"],
});
// 生成: useList + useDelete + keys
```

### 实际使用示例

```ts
// src/hooks/queries/resources/admins.ts
import { createResource } from "../create-resource";

const adminResource = createResource<
  PaginatedAdmin,
  Admin,
  CreateAdminInput,
  UpdateAdminInput,
  AdminQuery
>({
  resourceName: "admins",
  getClient: getClient as any,
  messages: {
    /* ... */
  },
});

// 导出 CRUD Hooks（按资源名重命名）
export const useAdmins = adminResource.useList; // 列表 Hook（复数形式）
export const useAdmin = adminResource.useDetail; // 详情 Hook（单数形式）
export const useCreateAdmin = adminResource.useCreate;
export const useUpdateAdmin = adminResource.useUpdate;
export const useDeleteAdmin = adminResource.useDelete;

// 扩展操作：手动编写 useMutation
export function useResetPassword() {
  const queryClient = useQueryClient();
  return useMutation({
    mutationFn: async ({ id, input }) => {
      const res = await getClient()[":id"]["reset-password"].$put({
        param: { id: String(id) },
        json: input,
      });
      return unwrapApiData(res, "重置密码失败");
    },
    onSuccess: (_, { id }) =>
      queryClient.invalidateQueries({
        queryKey: adminResource.keys.detail(id),
      }),
  });
}
```

### Query Keys 管理

`createResource` 内部会自动调用 `createQueryKeys` 生成规范且类型安全的 Query Keys：

- `keys.all`: `['resource']` — 全量清除
- `keys.list(params)`: `['resource', 'list', params]` — 列表查询
- `keys.detail(id)`: `['resource', 'detail', id]` — 详情查询

## 认证状态管理

```ts
// src/hooks/use-auth.ts — Zustand Store

interface AuthState {
  admin: AdminInfo | null;
  permissions: string[];
  menus: MenuItem[];
  initialized: boolean;
  loading: boolean;

  // Actions
  login(username, password): Promise<void>;
  logout(): void;
  refreshAuth(): Promise<void>;
  setInitialized(initialized: boolean): void;
}
```

当前实现中，浏览器请求默认依赖 `HttpOnly` 会话 Cookie，前端 Store 不持久化 session token。`createClient({ token })` 仅用于需要显式透传 Bearer Token 的非浏览器场景或测试场景。

**使用方式：**

```tsx
import { useAuth } from "@/hooks/use-auth";

function MyComponent() {
  const { admin, permissions, isAuthenticated, login, logout } = useAuth();
}
```

## UI 样式规范

项目基于 Mantine 7 进行开发，为保持视觉一致性，需遵循以下规范：

### Paper vs Card 选举原则

| 场景              | 选用组件 | 标准属性配置                                | 理由                                     |
| :---------------- | :------- | :------------------------------------------ | :--------------------------------------- |
| **搜索/过滤区域** | `Paper`  | `withBorder p="md" radius="md" mb="md"`     | 仅作为背景支撑，结构简单，无需 Card 功能 |
| **数据列表容器**  | `Card`   | `withBorder radius="md" p={0}`              | 逻辑权重高，支持 Card.Section 分割线     |
| **统计卡片**      | `Card`   | `withBorder p="lg" radius="lg"`             | 独立内容块，语义更准确，便于扩展         |
| **简单分组背景**  | `Paper`  | `p="sm" radius="md" bg="var(--mantine...)"` | 用于辅助视觉分割                         |

### 弹窗（Modal）内部规范

- **避免“框中框”**：Modal 本身已有边框和阴影，内部逻辑分组应**避免**使用 `withBorder` 的 Paper。
- **浅色背景分区**：推荐使用 `bg="var(--mantine-color-gray-0)"` 的 Paper 进行视觉分割，使界面更清盈。
- **统一圆角**：弹窗内部容器统一使用 `radius="md"`。

### 常用间距标准

- **页面级间距**：使用 `PageContainer` 自动处理，内部组件间距推荐使用 `Stack gap="xl"`。
- **搜索区与列表间距**：搜索容器必须设置 `mb="md"`。
- **内边距**：标准表单容器使用 `p="md"`，统计类使用 `p="lg"`。

### Card 全局悬停动效

项目通过 Mantine **Theme Styles API** + 全局 CSS 为所有 `Card` 组件统一注入悬停动效，无需在组件上手动添加 `className`：

```ts
// src/app/providers.tsx — createTheme 中注入 transition
Card: Card.extend({
  defaultProps: { shadow: 'sm', withBorder: true, radius: 'lg' },
  styles: {
    root: {
      transition: 'transform 0.2s cubic-bezier(0.4, 0, 0.2, 1), box-shadow 0.2s ...',
    },
  },
}),
```

```css
/* src/app/globals.css — 全局 hover 伪类 */
.mantine-Card-root:hover {
  transform: translateY(-2px);
  box-shadow: var(--app-card-hover-shadow);
}
```

如需**禁用特定 Card 的动效**，可在该 Card 上覆盖：`styles={{ root: { transition: 'none' } }}`。

## 主题配置

```ts
// src/app/providers.tsx
const theme = createTheme({
  primaryColor: "indigo",
  primaryShade: { light: 6, dark: 7 },
  fontFamily: "Inter, system-ui, -apple-system, sans-serif",
  defaultRadius: "md",
  components: {
    Card: Card.extend({
      defaultProps: { shadow: "sm", withBorder: true, radius: "lg" },
      styles: { root: { transition: "..." } }, // 全局悬停动效
    }),
    Paper: { defaultProps: { shadow: "xs", withBorder: true, radius: "lg" } },
    Modal: { defaultProps: { radius: "lg", overlayProps: { blur: 4 } } },
    // Button, TextInput, NavLink 等均有全局 transition 配置
  },
});
```

支持深色/浅色模式自动切换（`defaultColorScheme="auto"`）。
