# 数据库设计

## 技术选型

- **ORM**: Drizzle ORM 0.45（类型安全、Schema-first）
- **数据库**: MySQL 8.0+
- **驱动**: mysql2（连接池模式）
- **迁移**: Drizzle Kit（`drizzle-kit generate` / `drizzle-kit push`）

## Schema 详解

所有 Schema 文件位于 `src/db/schema/`，使用 `src/db/schema/index.ts` 统一导出。

### sys_admin — 管理员表

```ts
// src/db/schema/admin.ts
export const sysAdmin = mysqlTable("sys_admin", {
  id: bigint("id", { mode: "number", unsigned: true })
    .primaryKey()
    .autoincrement(),
  username: varchar("username", { length: 50 }).notNull(), // 唯一索引
  password: varchar("password", { length: 255 }).notNull(), // bcrypt 加密存储
  nickname: varchar("nickname", { length: 50 }).notNull().default(""),
  status: tinyint("status", { unsigned: true }).notNull().default(1), // 0-禁用 1-正常
  loginIp: varchar("login_ip", { length: 50 }), // 最后登录 IP
  loginTime: localDatetime("login_time"), // 最后登录时间
  remark: varchar("remark", { length: 500 }),
  createdAt: localDatetime("created_at")
    .notNull()
    .default(sql`CURRENT_TIMESTAMP`),
  updatedAt: localDatetime("updated_at")
    .notNull()
    .default(sql`CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP`),
});
// 索引: uk_username (UNIQUE), idx_status
```

### sys_role — 角色表

```ts
// src/db/schema/role.ts
export const sysRole = mysqlTable("sys_role", {
  id: bigint,
  roleName: varchar("role_name", { length: 50 }).notNull(),
  sort: int("sort", { unsigned: true }).notNull().default(0),
  status: tinyint, // 0-禁用 1-正常
  remark: varchar,
  createdAt,
  updatedAt,
});
// 索引: idx_status, idx_sort
```

### sys_menu — 菜单权限表

```ts
// src/db/schema/menu.ts
export const sysMenu = mysqlTable("sys_menu", {
  id: bigint,
  parentId: bigint("parent_id").notNull().default(0), // 0 表示顶级
  menuType: mysqlEnum("menu_type", ["D", "M", "B"]), // D-目录 M-菜单 B-按钮
  menuName: varchar("menu_name", { length: 50 }).notNull(),
  permission: varchar("permission", { length: 100 }), // 权限标识（如 system:admin:list）
  path: varchar("path", { length: 200 }), // 路由路径
  component: varchar("component", { length: 200 }), // 组件路径
  icon: varchar("icon", { length: 100 }), // Lucide 图标名
  sort: int,
  visible: tinyint, // 0-隐藏 1-显示
  status: tinyint, // 0-禁用 1-正常
  isExternal: tinyint, // 0-否 1-是（外链）
  isCache: tinyint, // 0-不缓存 1-缓存
  remark,
  createdAt,
  updatedAt,
});
// 索引: uk_permission (UNIQUE), idx_parent_id, idx_sort, idx_status, idx_menu_type
```

**菜单类型说明：**

| 类型 | 代码 | 说明                         | permission | path     | component |
| ---- | ---- | ---------------------------- | ---------- | -------- | --------- |
| 目录 | `D`  | 侧边栏目录节点               | -          | -        | -         |
| 菜单 | `M`  | 侧边栏菜单项（对应页面）     | 可选       | 路由路径 | 页面组件  |
| 按钮 | `B`  | 按钮级权限（不显示在侧边栏） | **必填**   | -        | -         |

### sys_operation_log — 操作日志表

记录增删改操作的审计日志，由 `auditLog` 中间件自动写入。

### sys_config — 系统配置表

键值对配置存储，支持 `string` / `boolean` / `number` / `json` / `array` 五种类型。

## Drizzle Relations

```ts
// src/db/schema/relations.ts
sysAdmin → many(sysAdminRole)          // 一个管理员有多个角色
sysRole  → many(sysAdminRole)          // 一个角色属于多个管理员
sysRole  → many(sysRoleMenu)           // 一个角色有多个菜单
sysMenu  → many(sysRoleMenu)           // 一个菜单属于多个角色
sysAdminRole → one(sysAdmin) + one(sysRole)  // 关联表回指
sysRoleMenu  → one(sysRole) + one(sysMenu)   // 关联表回指
```

## 数据库连接管理

```ts
// src/db/index.ts
const pool = mysql.createPool({
  uri: env.DATABASE_URL,
  connectionLimit: env.DATABASE_MAX_CONNECTIONS, // 默认 10
  idleTimeout: env.DATABASE_IDLE_TIMEOUT * 1000, // 默认 20s
  connectTimeout: env.DATABASE_CONNECT_TIMEOUT * 1000, // 默认 10s
  enableKeepAlive: true,
  timezone: "local",
});

export const db = drizzle({ client: pool, schema, mode: "default" });
```

**HMR 防护**：开发环境下将 pool 和 db 挂载到 `globalThis` 防止热更新时重复创建连接。

## 迁移工作流

```bash
# 开发环境：直接同步 Schema 到数据库（会自动生成/执行变更）
pnpm db:push

# 生产环境：生成迁移文件 → 审查 → 执行
pnpm db:generate   # 对比 Schema 与现有迁移，生成新的 SQL 文件到 drizzle/
pnpm db:migrate    # 执行未应用的迁移文件

# 启动时自动迁移（通过环境变量控制）
AUTO_DB_MIGRATE=true   # 应用启动时自动执行 drizzle/migrate
AUTO_DB_SEED=true      # 启动后自动运行种子
```

## Service 层数据库访问模式

```ts
// 查询示例 — src/server/services/system/admin/admin.query.ts
export async function findAdminByUsername(username: string) {
  return db.query.sysAdmin.findFirst({
    where: eq(sysAdmin.username, username),
  });
}

// 写入示例 — src/server/services/system/admin/admin.service.ts
export async function createAdmin(input: CreateAdminInput) {
  try {
    const hashed = await hashPassword(input.password);
    const [result] = await db
      .insert(sysAdmin)
      .values({ ...input, password: hashed });
    return getAdminById(Number(result.insertId));
  } catch (err) {
    throw handleDatabaseError(err); // 自动转换 MySQL 错误为业务错误
  }
}
```
