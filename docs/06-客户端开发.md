# 客户端开发

## 技术选型

| 职责       | 库                      | 说明                                      |
| ---------- | ----------------------- | ----------------------------------------- |
| UI 组件    | Mantine 7               | AppShell、Table、Modal、Form 等           |
| 服务端状态 | @tanstack/react-query 5 | API 数据加载、缓存、异步状态管理          |
| 客户端状态 | Zustand 5               | 认证 Store（token、用户信息、权限、菜单） |

| 通知       | Sonner                  | Toast 通知                                |

## Providers 架构

```ts
// src/app/providers.tsx
<MantineProvider>           // UI 组件主题
  <QueryClientProvider>      // React Query
    <AuthInitializer>        // 应用启动时刷新认证状态
      {children}
    </AuthInitializer>
    <Toaster />              // 全局 Toast
  </QueryClientProvider>
</MantineProvider>
```

**QueryClient 默认配置：**

```ts
queries: {
  staleTime: 5 * 60 * 1000,        // 5 分钟内数据不过期
  retry: 1,                        // 失败重试 1 次
  refetchOnWindowFocus: false,     // 窗口聚焦不自动刷新
},
mutations: { retry: 0 }            // mutation 不重试
```

## API Client（RPC Client）

```ts
// src/lib/client/index.ts

// 创建类型安全的 API 客户端
import { hc } from "hono/client";
import type { AppType } from "@/server/types";

export function getApiClient(): HonoClient {
  return hc<AppType>(`${baseUrl}/api`, {
    headers: () => ({
      "Content-Type": "application/json",
      Authorization: `Bearer ${getStoredToken()}`, // 自动从 localStorage 获取 token
    }),
  });
}

// 响应解包器 — 提取 data 字段，错误时抛出 Error
export async function unwrapApiData<T>(response, fallbackMessage): Promise<T> {
  if (!response.ok) {
    throw new Error(payload.message || fallbackMessage);
  }
  return payload.data as T;
}
```

**使用方式：**

```ts
// 直接调用（类型安全）
const client = getApiClient();
const res = await client.admins.$get({ query: { page: "1", pageSize: "20" } });
const data = await unwrapApiData(res, "获取管理员列表失败");
```

## React Query Hooks

### 工厂模式：createResource

`createResource` 是核心工厂函数，为每个 CRUD 资源自动生成 5 个 Hook：

```ts
// src/hooks/queries/core/create-resource.ts
const resource = createResource<TList, TDetail, TCreate, TUpdate, TQuery>({
  resourceName: 'admins',
  getClient: () => getApiClient().admins,
  defaultPageSize: 20,
  messages: {
    list: '获取管理员列表失败',
    detail: '获取管理员详情失败',
    create: '创建管理员失败',
    update: '更新管理员失败',
    delete: '删除管理员失败',
  },
})

// 生成的 Hooks：
resource.useList(params?)    // useQuery — 分页列表
resource.useDetail(id)       // useQuery — 单条详情
resource.useCreate()         // useMutation — 创建（成功后自动 invalidate）
resource.useUpdate()         // useMutation — 更新（成功后自动 invalidate）
resource.useDelete()         // useMutation — 删除（成功后自动 invalidate）
resource.keys                // Query Keys 对象
```

### 只读资源：createReadonlyResource

用于操作日志等只需查询 + 删除的资源：

```ts
const logResource = createReadonlyResource<TList, TQuery>({
  resourceName: "operation-logs",
  getClient: () => getApiClient()["operation-logs"],
});
// 生成: useList + useDelete + keys
```

### 实际使用示例

```ts
// src/hooks/queries/admins.ts
import { createResource } from "./core";

const adminResource = createResource<
  PaginatedAdmin,
  Admin,
  CreateAdminInput,
  UpdateAdminInput,
  AdminQuery
>({
  resourceName: "admins",
  getClient: getClient as any,
  messages: {
    /* ... */
  },
});

// 导出 CRUD Hooks
export const useAdminList = adminResource.useList;
export const useAdminDetail = adminResource.useDetail;
export const useCreateAdmin = adminResource.useCreate;
export const useUpdateAdmin = adminResource.useUpdate;
export const useDeleteAdmin = adminResource.useDelete;

// 扩展操作：手动编写 useMutation
export function useResetPassword() {
  const queryClient = useQueryClient();
  return useMutation({
    mutationFn: async ({ id, input }) => {
      const res = await getClient()["reset-password"][":id"].$put({
        param: { id: String(id) },
        json: input,
      });
      return unwrapApiData(res, "重置密码失败");
    },
    onSuccess: () =>
      queryClient.invalidateQueries({ queryKey: adminResource.keys.all }),
  });
}
```

### Query Keys 管理

```ts
// src/hooks/queries/core/query-keys.ts
export const queryKeys = {
  admins: createQueryKeys("admins"),
  roles: createQueryKeys("roles"),
  menus: createQueryKeys("menus"),
  configs: createQueryKeys("configs"),
  operationLogs: createQueryKeys("operation-logs"),
};

// 结构示例：
queryKeys.admins.all; // ['admins']
queryKeys.admins.lists(); // ['admins', 'list']
queryKeys.admins.list(params); // ['admins', 'list', { page: 1 }]
queryKeys.admins.details(); // ['admins', 'detail']
queryKeys.admins.detail(1); // ['admins', 'detail', 1]
```

## 认证状态管理

```ts
// src/hooks/use-auth.ts — Zustand Store

interface AuthState {
  token: string | null;
  admin: AdminInfo | null;
  permissions: string[];
  menus: MenuItem[];
  initialized: boolean;
  loading: boolean;

  // Actions
  login(username, password): Promise<void>;
  logout(): void;
  refreshAuth(): Promise<void>; // 使用现有 token 刷新用户信息
}

// 持久化：token 存储在 localStorage（key: 'auth-storage'）
```

**使用方式：**

```tsx
import { useAuth } from "@/hooks/use-auth";

function MyComponent() {
  const { admin, permissions, isAuthenticated, login, logout } = useAuth();
}
```
